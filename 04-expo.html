React Native com Expo Snack

Vamos consumir a API pública e apresentar as categorias no aplicativo.

Pré-requisito: a URL /categorias/api do Render precisa retornar JSON.
01

Crie um projeto Blank no Expo Snack

  1. Abra um novo Snack.
  2. Abra App.js.
  3. Substitua o código pelo exemplo abaixo.
02

Código do aplicativo

import React, { useEffect, useState } from "react";
import {
  View,
  Text,
  FlatList,
  StyleSheet,
  ActivityIndicator,
} from "react-native";

const API_URL =
  "https://SEU-PROJETO.onrender.com/categorias/api";

export default function App() {
  const [categorias, setCategorias] = useState([]);
  const [carregando, setCarregando] = useState(true);
  const [erro, setErro] = useState("");

  useEffect(() => {
    async function carregarCategorias() {
      try {
        const resposta = await fetch(API_URL);

        if (!resposta.ok) {
          throw new Error("Erro HTTP: " + resposta.status);
        }

        const dados = await resposta.json();

        console.log("Categorias:", dados);
        setCategorias(dados);
      } catch (error) {
        console.log("Erro:", error);
        setErro(error.message);
      } finally {
        setCarregando(false);
      }
    }

    carregarCategorias();
  }, []);

  if (carregando) {
    return (
      <View style={styles.container}>
        <ActivityIndicator size="large" />
        <Text style={styles.mensagem}>
          Carregando categorias...
        </Text>
      </View>
    );
  }

  if (erro) {
    return (
      <View style={styles.container}>
        <Text style={styles.titulo}>Erro</Text>
        <Text style={styles.erro}>
          Não foi possível carregar as categorias.
        </Text>
        <Text style={styles.detalhe}>{erro}</Text>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <Text style={styles.titulo}>Categorias</Text>
      <Text style={styles.subtitulo}>
        Categorias cadastradas
      </Text>

      <FlatList
        data={categorias}
        keyExtractor={(item) => item.id}
        contentContainerStyle={styles.lista}
        renderItem={({ item }) => (
          <View style={styles.card}>
            <Text style={styles.nome}>
              {item.nome}
            </Text>
            <Text style={styles.descricao}>
              {item.descricao}
            </Text>
          </View>
        )}
        ListEmptyComponent={
          <Text style={styles.vazio}>
            Nenhuma categoria cadastrada.
          </Text>
        }
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: "#f2f2f2",
    paddingTop: 50,
    paddingHorizontal: 20,
  },
  titulo: {
    fontSize: 28,
    fontWeight: "bold",
  },
  subtitulo: {
    fontSize: 16,
    color: "#666",
    marginTop: 5,
    marginBottom: 20,
  },
  lista: {
    paddingBottom: 20,
  },
  card: {
    backgroundColor: "#fff",
    padding: 18,
    marginBottom: 12,
    borderRadius: 10,
    borderWidth: 1,
    borderColor: "#ddd",
  },
  nome: {
    fontSize: 20,
    fontWeight: "bold",
    marginBottom: 8,
  },
  descricao: {
    fontSize: 16,
    color: "#555",
  },
  mensagem: {
    marginTop: 15,
    textAlign: "center",
    fontSize: 16,
  },
  erro: {
    fontSize: 17,
    color: "red",
    marginTop: 10,
  },
  detalhe: {
    marginTop: 10,
    color: "#666",
  },
  vazio: {
    textAlign: "center",
    marginTop: 30,
    color: "#777",
  },
});

04

Entenda o fluxo

useEffect()inicia a busca
→
fetch()chama a API
→
res.json()lê os dados
→
setCategorias()salva no estado
→
FlatListmostra na tela
05

Erros comuns

Network request failed
Confira a URL HTTPS do Render.
Erro 500
Confira o Firebase e a variável no Render.
Lista vazia
Confira se existem categorias no Firebase.
API não responde
Teste a URL no navegador antes do Snack.
